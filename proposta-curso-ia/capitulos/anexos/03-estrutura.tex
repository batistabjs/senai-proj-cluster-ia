\section{Infraestrutura Recomendada}

\textbf{Laboratório de Informática (uso geral)}

Adequado para:
\begin{itemize}
    \item Fundamentos da IA
    \item Programação Python
    \item Estatística
    \item Matemática
    \item Engenharia de Dados (básica)
    \item Engenharia de Prompt
    \item Engenharia de Contexto
    \item Desenvolvimento Web
    \item APIs
    \item Git
    \item Banco de Dados
    \item Projeto Integrador (parte documental)
\end{itemize}

Configuração mínima:
\begin{itemize}
    \item Intel Core i5 (12ª geração ou superior) ou AMD Ryzen 5 equivalente
    \item 16 GB RAM
    \item SSD de 512 GB
    \item Monitor Full HD
    \item Linux ou Windows 11
    \item Internet $\geq$ 300 Mbps
\end{itemize}

\textbf{Laboratório de Inteligência Artificial}

Necessário para:
\begin{itemize}
    \item Machine Learning
    \item Deep Learning
    \item IA Generativa
    \item LLMs
    \item LangGraph
    \item CrewAI
    \item Agentes Inteligentes
    \item Visão Computacional
    \item MLOps
\end{itemize}

Configuração recomendada:
\begin{itemize}
    \item Ryzen 9 ou Intel i9
    \item 64 GB RAM
    \item SSD NVMe 2 TB
    \item RTX 5080 16 GB (mínimo) ou superior
    \item Docker
    \item Kubernetes
    \item Ollama
    \item Open WebUI
    \item Cluster local de IA (opcional)
\end{itemize}

\textbf{Laboratório IoT / IIoT}
\begin{itemize}
    \item ESP32
    \item Raspberry Pi 5
    \item Arduino
    \item Sensores industriais
    \item Gateways MQTT
    \item OPC UA Server
    \item Switch Gerenciável
    \item Access Point Wi-Fi 6
    \item Servidor MQTT
    \item InfluxDB
    \item Grafana
    \item Node-RED
\end{itemize}
